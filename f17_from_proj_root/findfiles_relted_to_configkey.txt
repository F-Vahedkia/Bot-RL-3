"""
این خروجی به ما کمک می‌کند بفهمیم کدام فایل‌ها واقعاً configuration consumer هستند.
"""
Get-ChildItem -Path . -Recurse -File -Include *.py |
    Where-Object {
        $_.FullName -notmatch '\\(__pycache__|\.git|\.venv|venv)\\'
    } |
    Select-String -Pattern `
        '\bcfg\.get\(', `
        '\bself\.cfg\.get\(', `
        '\bconfig\.get\(', `
        '\bself\.config\.get\(', `
        '\bcfg\[', `
        '\bself\.cfg\[' |
    Select-Object Path, LineNumber, Line






"""
با این خروجی می‌توانیم ببینیم چه فایل‌هایی خود بخش risk را به‌عنوان یک block دریافت می‌کنند.
"""
Get-ChildItem -Path . -Recurse -File -Include *.py |
    Where-Object {
        $_.FullName -notmatch '\\(__pycache__|\.git|\.venv|venv)\\'
    } |
    Select-String -Pattern `
        'cfg\.get\(["'']risk', `
        'self\.cfg\.get\(["'']risk', `
        '\[["'']risk["'']\]' |
    Select-Object Path, LineNumber, Line
